# i18n Chinese/English (Three-Layer Full Translation) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement full i18n support for the token-usage CLI with Chinese/English switching, covering all three layers: command definitions, runtime output, and error messages.

**Architecture:** Self-built lightweight i18n package using `go:embed` for JSON translation files, a `T(key, args...)` lookup function, and language state management. Language detection follows a priority chain: `--lang` flag > `TOKEN_USAGE_LANG` env > `config.yaml language` > system `LANG`/`LC_ALL` > default `en`. A new `tu lang` command persists language choice to config.

**Tech Stack:** Go 1.26.6, cobra, `go:embed`, `golang.org/x/text` (NOT needed - we use `go-runewidth` already in deps for CJK alignment), standard library `testing`.

**Spec:** This plan implements the spec in the user's task description (i18n three-layer full translation for token-usage CLI).

## Global Constraints

- Go 1.26.6 (no newer features required)
- No new third-party dependencies (use `go:embed` + existing `go-runewidth`)
- `en.json` is the single source of truth for all keys
- `zh.json` missing keys fall back to English
- Key consistency test must enforce en/zh key parity
- JSON keys are NEVER translated
- Command names, flag names, aliases, completion help, "Error:" prefix, HTTP status codes are NOT translated
- Chinese translations keep commands/flags in English (e.g. `token-usage account switch`)
- Chinese面向开发者简洁中文语域
- Pre-commit: gofmt + go vet
- Single commit with `feat(i18n)` prefix
- All existing tests must pass

```go
package i18n

import (
    "embed"
    "fmt"
    "os"
    "strings"
    "sync"
)

//go:embed en.json
var enFS embed.FS

//go:embed zh.json
var zhFS embed.FS

var (
    mu       sync.RWMutex
    lang     = "en"
    enDict   map[string]string
    zhDict   map[string]string
    once     sync.Once
)

func loadDictionaries() {
    once.Do(func() {
        enDict = loadJSON(enFS, "en.json")
        zhDict = loadJSON(zhFS, "zh.json")
    })
}

func loadJSON(fs embed.FS, name string) map[string]string {
    data, err := fs.ReadFile(name)
    if err != nil {
        return map[string]string{}
    }
    var m map[string]string
    if err := json.Unmarshal(data, &m); err != nil {
        return map[string]string{}
    }
    return m
}

func SetLanguage(l string) {
    mu.Lock()
    defer mu.Unlock()
    if l == "zh" || l == "en" {
        lang = l
    }
}

func GetLanguage() string {
    mu.RLock()
    defer mu.RUnlock()
    return lang
}

func T(key string, args ...any) string {
    loadDictionaries()
    mu.RLock()
    currentLang := lang
    mu.RUnlock()

    var val string
    var ok bool
    if currentLang == "zh" {
        val, ok = zhDict[key]
    }
    if !ok {
        val, ok = enDict[key]
    }
    if !ok {
        return key
    }
    if len(args) > 0 {
        return fmt.Sprintf(val, args...)
    }
    return val
}

// DetectLanguage resolves language from priority chain:
// flag > env > config > systemLang > default "en"
func DetectLanguage(flagVal, envVal, configVal, systemLang string) string {
    if flagVal != "" {
        return normalizeLang(flagVal)
    }
    if envVal != "" {
        return normalizeLang(envVal)
    }
    if configVal != "" {
        return normalizeLang(configVal)
    }
    if strings.HasPrefix(systemLang, "zh") {
        return "zh"
    }
    return "en"
}

func normalizeLang(s string) string {
    s = strings.ToLower(strings.TrimSpace(s))
    if strings.HasPrefix(s, "zh") {
        return "zh"
    }
    return "en"
}
```

```go
package cmd

import (
    "fmt"
    "github.com/emmmdty/token-usage/internal/config"
    "github.com/emmmdty/token-usage/internal/i18n"
    "github.com/spf13/cobra"
)

var langCmd = &cobra.Command{
    Use:     "lang [zh|en]",
    Aliases: []string{"language"},
    Short:   i18n.T("cmd.lang.short"),
    Long:    i18n.T("cmd.lang.long"),
    Args:    cobra.MaximumNArgs(1),
    RunE: func(cmd *cobra.Command, args []string) error {
        if len(args) == 0 {
            current := i18n.GetLanguage()
            fmt.Printf("Current language: %s\n", current)
            return nil
        }
        newLang := args[0]
        if newLang != "zh" && newLang != "en" {
            return fmt.Errorf("unsupported language: %s (use 'zh' or 'en')", newLang)
        }
        cfgPath, err := getConfigPath()
        if err != nil {
            return err
        }
        cfg, err := config.LoadOrCreateConfig(cfgPath)
        if err != nil {
            return err
        }
        cfg.Language = newLang
        if err := config.SaveConfig(cfg, cfgPath); err != nil {
            return err
        }
        i18n.SetLanguage(newLang)
        // Bilingual confirmation
        if newLang == "zh" {
            fmt.Println("语言已切换为中文")
            fmt.Println("Language switched to Chinese")
        } else {
            fmt.Println("Language switched to English")
            fmt.Println("语言已切换为英文")
        }
        return nil
    },
}

func init() {
    rootCmd.AddCommand(langCmd)
}
```
